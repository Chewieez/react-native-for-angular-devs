Lesson 0013: Native Camera & Image Picker

Translating HTML5 file inputs and MediaDevices to expo-camera (CameraView), expo-image-picker, and device permission hooks

1. Web File Uploads vs. Mobile Hardware Access

In Angular web applications, image selection is done via <input type="file"> or WebRTC navigator.mediaDevices.getUserMedia(). On mobile devices, operating systems require strict runtime permission prompts and native hardware bridges:

Concept Angular (Web) React Native (Mobile)
Photo Library <input type="file" accept="image/*"> ImagePicker.launchImageLibraryAsync() (Native Photo Picker)
Camera Viewfinder <video> + WebRTC streams <CameraView /> from expo-camera (Native AVFoundation / CameraX)
Permissions Browser popup on call useCameraPermissions() hook + app.json native config

2. Photo Library Picker (expo-image-picker)

The standard way to select an existing photo or prompt the user for an avatar or post upload:

// src/utils/imagePicker.ts
import * as ImagePicker from 'expo-image-picker';

export async function pickImage(): Promise<string | null> {
  // 1. Request library permissions
  const { status } = await ImagePicker.requestMediaLibraryPermissionsAsync();
  if (status !== 'granted') {
    alert('Permission to access photos is required!');
    return null;
  }

  // 2. Launch native image picker
  const result = await ImagePicker.launchImageLibraryAsync({
    mediaTypes: ImagePicker.MediaTypeOptions.Images,
    allowsEditing: true,
    aspect: [1, 1], // Square crop (e.g. BeReal avatar)
    quality: 0.8,   // Compress image to save mobile bandwidth
  });

  if (!result.canceled && result.assets && result.assets.length > 0) {
    return result.assets[0].uri; // Local file URI (e.g. file:///data/.../photo.jpg)
  }

  return null;
}

3. Native Viewfinder with expo-camera (Modern CameraView)

For custom in-app camera viewfinders (like BeReal or Instagram Stories), use CameraView with the useCameraPermissions() hook:

// src/components/NativeCameraView.tsx
import React, { useRef, useState } from 'react';
import { View, Text, TouchableOpacity, StyleSheet } from 'react-native';
import { CameraView, useCameraPermissions, CameraType } from 'expo-camera';

export default function NativeCameraView({ onPhotoCaptured }: { onPhotoCaptured: (uri: string) => void }) {
  const [facing, setFacing] = useState<CameraType>('back');
  const [permission, requestPermission] = useCameraPermissions();
  const cameraRef = useRef<CameraView>(null);

  if (!permission) {
    // Permissions are still loading
    return <View />;
  }

  if (!permission.granted) {
    return (
      <View style={styles.container}>
        <Text style={styles.text}>We need camera permission to take your daily post</Text>
        <TouchableOpacity style={styles.button} onPress={requestPermission}>
          <Text style={styles.buttonText}>Grant Permission</Text>
        </TouchableOpacity>
      </View>
    );
  }

  const takePicture = async () => {
    if (cameraRef.current) {
      const photo = await cameraRef.current.takePictureAsync({ quality: 0.8 });
      if (photo?.uri) {
        onPhotoCaptured(photo.uri);
      }
    }
  };

  return (
    <View style={styles.container}>
      <CameraView style={styles.camera} facing={facing} ref={cameraRef}>
        <View style={styles.overlay}>
          <TouchableOpacity style={styles.captureBtn} onPress={takePicture}>
            <View style={styles.innerCircle} />
          </TouchableOpacity>
        </View>
      </CameraView>
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, backgroundColor: '#000' },
  camera: { flex: 1 },
  overlay: { flex: 1, justifyContent: 'flex-end', alignItems: 'center', marginBottom: 40 },
  captureBtn: { width: 74, height: 74, borderRadius: 37, borderWidth: 4, borderColor: '#fff', justifyContent: 'center', alignItems: 'center' },
  innerCircle: { width: 58, height: 58, borderRadius: 29, backgroundColor: '#fff' },
  text: { color: '#fff', textAlign: 'center', marginBottom: 20 },
  button: { backgroundColor: '#007AFF', padding: 12, borderRadius: 8, alignSelf: 'center' },
  buttonText: { color: '#fff', fontWeight: 'bold' }
});

4. Uploading Local Native URIs to Backend Storage

On mobile, selected files exist as local device paths (e.g. file:///var/mobile/.../temp.jpg). To upload to Supabase or Firebase, fetch the file URI using expo-file-system and decode it via base64-arraybuffer as an ArrayBuffer:

// src/lib/supabase/storage.ts
import { supabase } from './client';
import * as FileSystem from 'expo-file-system';
import { decode } from 'base64-arraybuffer';

export async function uploadPostImage(localUri: string, userId: string): Promise<string> {
  // Read local file as base64 string
  const base64 = await FileSystem.readAsStringAsync(localUri, {
    encoding: FileSystem.EncodingType.Base64,
  });

  const filePath = `${userId}/${Date.now()}.jpg`;
  const contentType = 'image/jpeg';

  const { error } = await supabase.storage
    .from('posts')
    .upload(filePath, decode(base64), { contentType });

  if (error) throw error;

  const { data } = supabase.storage.from('posts').getPublicUrl(filePath);
  return data.publicUrl;
}
